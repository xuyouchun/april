#
# CMake build for April.
#
# Coexists with the legacy Makefile system (Makefile / Makefile.rule /
# Makefile.options). Typical usage:
#
#   cmake -B build -DCMAKE_BUILD_TYPE=Debug
#   cmake --build build -j
#
# Executables go to bin/ (same location as the legacy build), so ./run and
# test/unittest keep working unchanged.
#

cmake_minimum_required(VERSION 3.16)

project(april LANGUAGES CXX)

#---------- ---------- ---------- ---------- ----------
# Options (mirror Makefile.options)

set(APRIL_ARCH   "x86_64"  CACHE STRING "Architecture (subdirectory of arch/)")
set(APRIL_OS     "linux"   CACHE STRING "Operating system (subdirectory of os/)")
set(APRIL_GLOBAL "default" CACHE STRING "Global configuration (subdirectory of global/)")

set(CONFIG_BASE_NAMESPACE    "april" CACHE STRING "Root C++ namespace")
set(CONFIG_NULLPTR_SUPPORTED 1       CACHE STRING "nullptr supported")

#---------- ---------- ---------- ---------- ----------
# Global compile settings (mirror Makefile.rule)

set(CMAKE_CXX_STANDARD 20)          # legacy: -std=c++2a
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    # Legacy default was -O0; Debug (-g -O0) is the closest equivalent.
    set(CMAKE_BUILD_TYPE "Debug" CACHE STRING "Build type" FORCE)
endif()

# CONFIG_OPTIMIZE for config.h (legacy: OPTIMIZE=<0|1|2|3>).
#
# CONFIG_OPTIMIZE != 0 turns X_DEBUG off (noexcept, no assertions, unchecked casts); that
# configuration crashes at run time, so every build type keeps CONFIG_OPTIMIZE at 0: the
# compiler still optimizes (Release: -O3), only the run-time checks stay. Set
# APRIL_CONFIG_OPTIMIZE (0..3) to override.
set(CONFIG_OPTIMIZE 0)

set(APRIL_CONFIG_OPTIMIZE "" CACHE STRING "Override CONFIG_OPTIMIZE of config.h (0..3)")
if(NOT APRIL_CONFIG_OPTIMIZE STREQUAL "")
    set(CONFIG_OPTIMIZE ${APRIL_CONFIG_OPTIMIZE})
endif()

# Generated config.h (legacy: written by Makefile.rule into include/config.h)
configure_file(cmake/config.h.in "${CMAKE_BINARY_DIR}/generated/config.h")

# Stale files from the legacy build would shadow the CMake-generated ones
# (quoted includes and -I search order both prefer the source tree).
foreach(__stale "include/config.h" "modules/lang_cs/text.csec")
    if(EXISTS "${CMAKE_SOURCE_DIR}/${__stale}")
        message(WARNING
            "${__stale} exists (generated by the legacy Makefile build) and "
            "will shadow the CMake-generated one. Run 'make distclean' or "
            "delete it before building with CMake.")
    endif()
endforeach()

# Executables go to bin/ like the legacy build; test/unittest expects bin/april.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")

#---------- ---------- ---------- ---------- ----------
# Dependencies

find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS log system filesystem)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

#---------- ---------- ---------- ---------- ----------
# Common build interface shared by every module

add_library(april_options INTERFACE)

# Legacy: INC_PATH + BASE_PATH + OS/ARCH/GLOBAL paths.
target_include_directories(april_options INTERFACE
    "${CMAKE_SOURCE_DIR}/include"
    "${CMAKE_SOURCE_DIR}/include/modules"
    "${CMAKE_SOURCE_DIR}"
    "${CMAKE_SOURCE_DIR}/os/${APRIL_OS}"
    "${CMAKE_SOURCE_DIR}/arch/${APRIL_ARCH}"
    "${CMAKE_SOURCE_DIR}/global/${APRIL_GLOBAL}"
    "${CMAKE_BINARY_DIR}/generated"
)

# MinGW defaults to the MSVC bit-field layout, which does not pack adjacent bit-fields of
# different types into one storage unit (see analyze_node_key_t). Use the Itanium layout.
if(WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(april_options INTERFACE -mno-ms-bitfields)
endif()

# The code type-puns (e.g. analyze_node_key_t -> its underlying integer).
target_compile_options(april_options INTERFACE -fno-strict-aliasing)

if(NOT WIN32)
    target_compile_definitions(april_options INTERFACE
        BOOST_LOG_DYN_LINK
        BOOST_FILESYSTEM_DYN_LINK
    )
endif()

target_link_libraries(april_options INTERFACE
    Boost::log
    Boost::system
    Boost::filesystem
    Threads::Threads
)

#---------- ---------- ---------- ---------- ----------
# Modules: april_add_module(<name> <source dir>) -> static library

function(april_add_module NAME DIR)
    file(GLOB __sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/${DIR}/*.cpp")
    add_library(${NAME} STATIC ${__sources})
    target_link_libraries(${NAME} PUBLIC april_options)
endfunction()

# Kernel modules (legacy: KMODULES)
april_add_module(global    "global/${APRIL_GLOBAL}")
april_add_module(common    "common")
april_add_module(algorithm "algorithm")
april_add_module(arch      "arch/${APRIL_ARCH}")
april_add_module(os        "os/${APRIL_OS}")
april_add_module(lib       "lib")

# Modules (legacy: MODULES, minus the april executable itself)
april_add_module(mm      "modules/mm")
april_add_module(compile "modules/compile")
april_add_module(core    "modules/core")
april_add_module(lang_cs "modules/lang_cs")
april_add_module(lang_lq "modules/lang_lq")
april_add_module(rtlib   "modules/rtlib")
april_add_module(exec    "modules/exec")
april_add_module(rt      "modules/rt")
april_add_module(native  "modules/native")   # LLVM IR back end + native runtime support

# The directory of the static libraries, where the executables of the native programs are linked
# with them: it is searched when nothing else tells where they are.
target_compile_definitions(native PRIVATE APRIL_BUILD_LIBS_DIR="${CMAKE_BINARY_DIR}")

# Same archive order as the legacy link line; wrapped in --start-group /
# --end-group on Linux so cross-archive references never depend on ordering.
set(APRIL_MODULE_LIBS
    global common algorithm arch os lib
    mm compile core lang_cs lang_lq rtlib exec rt native
)

function(april_link_modules TARGET)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(${TARGET} PRIVATE
            -Wl,--start-group ${APRIL_MODULE_LIBS} -Wl,--end-group)
    else()
        target_link_libraries(${TARGET} PRIVATE ${APRIL_MODULE_LIBS})
    endif()
endfunction()

#---------- ---------- ---------- ---------- ----------
# lang_cs: text.src -> text.csec (legacy: tools/csec rule in Makefile.rule)

set(__csec_dir  "${CMAKE_BINARY_DIR}/generated/lang_cs")
set(__csec_file "${__csec_dir}/text.csec")

add_custom_command(
    OUTPUT "${__csec_file}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${__csec_dir}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/text2cpp.py"
            "${CMAKE_SOURCE_DIR}/modules/lang_cs/text.src" "${__csec_file}"
    DEPENDS "${CMAKE_SOURCE_DIR}/modules/lang_cs/text.src"
            "${CMAKE_SOURCE_DIR}/tools/text2cpp.py"
    COMMENT "Generating lang_cs/text.csec"
    VERBATIM
)

target_sources(lang_cs PRIVATE "${__csec_file}")
set_source_files_properties("${__csec_file}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_include_directories(lang_cs PRIVATE "${__csec_dir}")

#---------- ---------- ---------- ---------- ----------
# Executables

# april (legacy: modules/april, TARGET_TYPE=execute)
file(GLOB __april_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/modules/april/*.cpp")
add_executable(april ${__april_sources})
target_link_libraries(april PRIVATE april_options)
april_link_modules(april)

# test/default (legacy: `./run` -> bin/default); built on demand:
#   cmake --build build --target test_default
file(GLOB __test_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/test/default/*.cpp")
add_executable(test_default EXCLUDE_FROM_ALL ${__test_sources})
set_target_properties(test_default PROPERTIES OUTPUT_NAME "default")
target_link_libraries(test_default PRIVATE april_options)
april_link_modules(test_default)

#---------- ---------- ---------- ---------- ----------
# Tests: ctest runs the existing pytest suite.
# Requires bin/april (built above) and System.apl at the repository root.

enable_testing()

add_test(
    NAME unittest
    COMMAND ${Python3_EXECUTABLE} -m pytest .
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}/test/unittest"
)
